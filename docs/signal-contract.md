# Standard Signal Contract Specification

## 1. Overview
The platform defines a strict, standardized JSON signal contract for all trading signals generated by any strategy module. Signals must never be generated using future candles or non-causal data.

---

## 2. Formal JSON Signal Schema

```json
{
  "$schema": "https://json-schema.org/draft/2020-12/schema",
  "title": "AlphaQuantStandardSignal",
  "type": "object",
  "required": [
    "signal_id",
    "created_at",
    "data_timestamp",
    "exchange",
    "symbol",
    "timeframe",
    "side",
    "entry_zone",
    "stop_loss",
    "take_profit_levels",
    "risk_reward_ratio",
    "confidence",
    "strategy_name",
    "market_regime",
    "indicators",
    "reasons",
    "invalidation_conditions",
    "expires_at",
    "status"
  ],
  "properties": {
    "signal_id": {"type": "string"},
    "created_at": {"type": "string", "format": "date-time"},
    "data_timestamp": {"type": "string", "format": "date-time"},
    "exchange": {"type": "string"},
    "symbol": {"type": "string"},
    "timeframe": {"type": "string"},
    "side": {"type": "string", "enum": ["LONG", "SHORT", "NEUTRAL"]},
    "entry_zone": {
      "type": "object",
      "required": ["min", "max"],
      "properties": {
        "min": {"type": "number"},
        "max": {"type": "number"}
      }
    },
    "stop_loss": {"type": "number"},
    "take_profit_levels": {
      "type": "array",
      "items": {"type": "number"},
      "minItems": 1
    },
    "risk_reward_ratio": {"type": "number", "minimum": 1.0},
    "confidence": {"type": "number", "minimum": 0.0, "maximum": 100.0},
    "strategy_name": {"type": "string"},
    "market_regime": {"type": "string", "enum": ["TREND", "RANGE", "VOLATILE", "CHOPPY", "UNKNOWN"]},
    "indicators": {"type": "object"},
    "reasons": {"type": "array", "items": {"type": "string"}},
    "invalidation_conditions": {"type": "array", "items": {"type": "string"}},
    "expires_at": {"type": "string", "format": "date-time"},
    "status": {"type": "string", "enum": ["candidate", "approved", "rejected", "expired"]}
  }
}
```
